жёлтым = 6 слов, общих для piece0 и piece1.

Псевдокод chunk_markdown (NEW) 
page_title = first_h1(content) or humanize(basename)        # до первого ##
for section in split_on_h2(content):                        # ^##\s+
    heading = section.heading
    body    = section.body                                  # вкл. ###-подзаголовки
    lead    = first_paragraph(body)[:250]
    prefix  = f"# {page_title}\n## {heading}\n{lead}".strip()
    for ci, piece in enumerate(split_words(body, size, overlap)):
        text = prefix + "\n\n" + " ".join(piece)
        yield Chunk(file, heading, ci, text, hash=sha256(text)[:16])
